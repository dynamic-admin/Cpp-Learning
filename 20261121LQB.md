#include <bits/stdc++.h>
using namespace std;
//判断是否是质数
bool prime(int n){
    if(n<=1) return false;
    if(n==2) return true;
    if(n%2==0&&n!=2) return false;//很重要
    if(n>=3){
        for(int i=3;i*i<=n;i+=2){
            if(n%i==0) return false;
        }
    }
    return true;
}

int main(){
    int n;
    cin >>n;
    for(int i=4;i<=n;i+=2){
        for(int j=2;j<=n/2;j++){
            if(prime(j)&&prime(i-j)){
                cout <<i<<'='<<j<<'+'<<i-j<<endl;
                break;
            }
        }
    }
    return 0;
}
