#include <bits/stdc++.h>
using namespace std;
//判断是否是质数
bool prime(int n){
    if(n<=1) return false;
    if(n==2) return true;
    if(n%2==0&&n!=2) return false;
    if(n>=3){
        for(int i=3;i*i<=n;i+=2){
            if(n%i==0) return false;
        }
    }
    return true;
}

int main(){
    int n;
    cin >>n;
    for(int i=2;i*i<=n;i++){
        if(n%i==0){//一开始把这个忘记了，这就导致不过
            if(prime(i)&&prime(n/i)){
                cout <<n/i<<endl;
                break;
            }
        }
    }
    return 0;
}
